agregarFoto :: RV -> Foto -> RV
agregarFoto (MkRV mf mv) (MkF c vehs) =
  case lookupM c mf of
    Just _  -> MkRV mf mv  -- ya existe la foto, no se modifica
    Nothing ->
      let nuevoMF = assocM c (MkF c vehs) mf
          listaVehs = set2list vehs
          nuevoMV = actualizarCoapariciones listaVehs mv
      in MkRV nuevoMF nuevoMV


-- Actualiza el mapa de coapariciones, recursivamente
actualizarCoapariciones :: [Vehiculo] -> Map Vehiculo (Set Vehiculo) -> Map Vehiculo (Set Vehiculo)
actualizarCoapariciones [] mv = mv
actualizarCoapariciones (v:vs) mv =
  let mv'  = agregarPares v vs mv
  in actualizarCoapariciones vs mv'

-- Agrega simétricamente los pares (v, v2) para cada v2 en la lista dada
agregarPares :: Vehiculo -> [Vehiculo] -> Map Vehiculo (Set Vehiculo) -> Map Vehiculo (Set Vehiculo)
agregarPares _ [] mv = mv
agregarPares v (v2:resto) mv =
  let mv'  = agregarPar v v2 mv
  in agregarPares v resto mv'

-- Agrega relación simétrica v1-v2
agregarPar :: Vehiculo -> Vehiculo -> Map Vehiculo (Set Vehiculo) -> Map Vehiculo (Set Vehiculo)
agregarPar v1 v2 mv =
  let set1 = case lookupM v1 mv of
               Nothing -> addS v2 emptyS
               Just s  -> addS v2 s
      mv1  = assocM v1 set1 mv
      set2 = case lookupM v2 mv1 of
               Nothing -> addS v1 emptyS
               Just s  -> addS v1 s
  in assocM v2 set2 mv1

Para una foto con vf vehículos:

actualizarCoapariciones recorre la lista → O(vf)

Cada llamada a agregarPares recorre los demás vehículos → O(vf)

En cada par se hacen operaciones sobre Map y Set → O(log vf)

Total: O(vf² * log vf), tal como exige el enunciado.

quienesAparecen :: RV -> CUIF -> Set Vehiculo

				 Nothing -> error "El CUIF no corresponde a una foto de RV" 
				 Just (MkF c sv) -> sv

aparecenJuntos :: RV -> Vehiculo -> Vehiculo -> Bool
aparecenJuntos (MkRV _ mvs) v1 v2 = if v1 /= v2 
				    then case lookupM v1 mv of
    					      Nothing   -> False
                                              Just setV -> belongsS v2 setV
			            else error "Los vehículos tienen que ser diferentes"

data RV = MkRV (Map CUIF Foto)              -- Fotos del registro
               (Map Vehiculo (Set Vehiculo))-- Coapariciones
               (Heap (Int, CUIF))           -- Heap ordenada por cantidad de vehículos (máximo arriba)

masCargada :: RV -> Maybe CUIF
masCargada (MkRV _ _ heap) = if isEmptyH heap 
			     then Nothing
                             else Just (snd (findMaxH heap))
